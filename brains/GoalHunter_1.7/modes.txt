# GoalHunter modes.
#
# This file is the brain's own answer to "which modes can I be run in, and
# which difficulty levels does each of them have?". The lobby reads it
# locally on every machine (brains ship with the client), fills the gear
# popup's Mode and Difficulty dropdowns from it, and hands the chosen KEYS
# to the brain as the "mode=" and "difficulty=" BRAIN_INIT_ARG tokens.
#
# Section header = the mode key. First section is mode 0, the one every
# ordinary game uses. "levels" is a comma-separated list of key:Label, and
# "default" names the level key a freshly added bot starts at.
#
# Two optional lines per section:
#   about = <one line>   shown under the lobby's Mode combo and as the combo
#                        item's hover text. Plain text, not translated, cut
#                        to 159 bytes. No line = nothing shown. A "#"
#                        in it starts a comment.
#   standard_levels = yes   this mode's easy/medium/hard mean the same as
#                        default's, so the lobby shows its own translated
#                        Easy/Medium/Hard names and descriptions for them.
#                        Without it only [default] gets that wording.
#
# The brain applies the level in constants.lua (M.MODE_LEVELS[mode][level],
# a bundle of knob overrides applied before any preset= or cfg= token):
# hard is an empty bundle, so Hard is today's brain unchanged; medium and
# easy hand the bot handicaps. The survival mode's bundles are still empty.
# The turtle mode's bundles are default's with PILL_PLACE_TURTLE turned on.
#
# open_default (above the first section) names the mode a new bot starts in
# when the game type is Open. Tournament, Strict and scripted games start
# in mode 0.
# A mode picked by hand in the lobby, and a scenario's own mode, still win.

open_default = turtle

[default]
label = Default
levels = easy:Easy:1, medium:Medium:2, hard:Hard:3
default = hard

# Survival. Its real levels have not been decided — these three are
# placeholders so the dropdown is wired end to end; edit them here and
# nothing else has to change. The label is what the lobby shows on the
# row's mode tag, so it is kept to one short word.
[survival]
label = Survival
levels = easy:Easy:1, medium:Medium:2, hard:Hard:3
default = hard

# Turtle. The same three levels as default; each one is default's level plus
# PILL_PLACE_TURTLE (the bot builds one tight, defended cluster of pills
# around its home instead of spreading them out). constants.lua builds its
# bundles from default's, so the two cannot drift, and standard_levels
# lets the lobby describe them in default's words.
[turtle]
label = Turtle
about = Builds its pillboxes close together in one cluster at home instead of spreading them out. Otherwise it plays like Default.
levels = easy:Easy:1, medium:Medium:2, hard:Hard:3
default = hard
standard_levels = yes
